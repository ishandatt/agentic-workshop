# Pinned to compatible ranges, not bare floors.
#
# The previous floors were fiction: `langchain>=0.3` resolved to 1.3.14, a major
# version past the floor, so a fresh install months apart could land somewhere
# very different. Each line below is >= the version this workshop was built and
# verified against, and < the next major, so `pip install -r` reproduces a
# working stack without freezing it forever.

# Core — the LangChain family moves together; pin them as a set.
langchain>=1.3.14,<2
langchain-core>=1.5.3,<2         # imported directly: messages, prompts, LCEL
langchain-text-splitters>=1.1.2,<2   # imported directly: chunking for RAG
langchain-ollama>=1.1.0,<2
langchain-postgres>=0.0.17,<0.1
langgraph>=1.2.10,<2
langgraph-checkpoint-postgres>=3.1.1,<4   # module 8: durable interrupts

# Service
fastapi>=0.141.1,<1
uvicorn>=0.52,<1
httpx>=0.28,<1
pydantic>=2.13,<3

# MCP
# Pinned below 2.0 deliberately. mcp 2.0.0 landed 2026-07-28 and is a breaking
# rewrite: `mcp.server.fastmcp` is gone (replaced by `mcp.server.mcpserver`),
# and response fields moved from camelCase to snake_case (inputSchema ->
# input_schema). Nearly all published examples — and langchain-mcp-adapters,
# which imports a symbol 2.0 removed — still target 1.x. Unpinned, `mcp>=1.1`
# silently resolves to 2.0 and module 3 breaks in ways that look like your fault.
mcp>=1.29,<2

# Database
psycopg[binary]>=3.3,<4
# The `pgvector` Python package is deliberately not listed: no code imports it,
# and langchain-postgres already depends on it. That dependency caps it at
# <0.4, so pip resolves 0.3.6 — an IDE reporting "outdated, latest is 0.5.0" is
# wrong; requiring 0.5 makes the install unresolvable. Add a line here only if
# something actually does `import pgvector`, and match langchain-postgres's cap.

# Bonus modules 13-16
# NOTE: install google-adk WITHOUT the [extensions] extra. That extra pins
# langgraph>=0.2.60,<0.4.8, which is incompatible with langchain 1.x and
# silently breaks modules 3, 8 and 9.
google-adk>=2.6.1,<3
litellm>=1.95,<2
mlx-lm>=0.31,<1          # module 16 only; Apple Silicon

# Testing
pytest>=9.1,<10          # module 6 runs the eval suite as tests

# Utilities
rich>=13.9,<15
python-dotenv>=1.2,<2
